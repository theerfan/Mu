cmake_minimum_required(VERSION 3.10)
project(gdsp CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_library(gdsp STATIC src/gdsp.cpp)
target_include_directories(gdsp PUBLIC include)
add_executable(gtcli tools/gtcli.cpp)
target_link_libraries(gtcli gdsp)
add_executable(gen_test_wavs tools/gen_test_wavs.cpp)
target_include_directories(gen_test_wavs PRIVATE include)
add_executable(bench tools/bench.cpp)
target_link_libraries(bench gdsp)
