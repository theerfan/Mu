# mucore: platform-independent practice core (scoring, tuner, calibration math, SPSC queues), shared by
# Android (NDK) and later iOS. Depends only on gdsp (../dsp) and the C++17 standard library.
cmake_minimum_required(VERSION 3.10)
project(mucore CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
if(NOT TARGET gdsp)
  set(GDSP_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
  add_subdirectory(${CMAKE_CURRENT_LIST_DIR}/../dsp ${CMAKE_CURRENT_BINARY_DIR}/gdsp)
endif()
add_library(mucore STATIC src/practice.cpp src/synth.cpp)
target_include_directories(mucore PUBLIC include)
target_link_libraries(mucore PUBLIC gdsp)
option(MU_BUILD_TESTS "Build host tests" ON)
if(MU_BUILD_TESTS)
  add_executable(mucore_tests test/test_core.cpp)
  target_link_libraries(mucore_tests mucore)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Android")
    find_package(Threads REQUIRED)
    target_link_libraries(mucore_tests Threads::Threads)
  endif()
endif()
